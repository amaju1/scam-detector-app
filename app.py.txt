# 2. MOBILE INFERENCE CONTROL
# ==========================================
def analyze_mobile_text(text):
    time.sleep(0.8) # Quick response simulation for mobile networks
    text_lower = text.lower()
    score = 0.04
    flags = []
    
    if any(w in text_lower for w in ["urgent", "now", "verify", "suspended"]):
        score += 0.40
        flags.append("URGENCY ATTACK VECTOR")
    if any(w in text_lower for w in ["link", "click", "http", "secure"]):
        score += 0.35
        flags.append("OUTBOUND CALL-TO-ACTION")
        
    return min(score, 0.99), flags
